💻 Standalone Three.js Tank Implementation

Here is the complete, self-contained Three.js code that builds this exact 3D assembly (matching Image 1 & Image 2). You can drop this into any Three.js project, vanilla HTML file, React Three Fiber, or Vite project! 

import * as THREE from 'three';
import { OrbitControls } from 'three/examples/jsm/controls/OrbitControls.js';

// 1. Scene, Camera, Renderer
const scene = new THREE.Scene();
scene.background = new THREE.Color(0x9aa8b8); // Authentic CAD studio grey-blue

const camera = new THREE.PerspectiveCamera(40, window.innerWidth / window.innerHeight, 0.1, 100);
// Camera angle matching Image 1:
camera.position.set(7.8, 6.4, 9.8);

const renderer = new THREE.WebGLRenderer({ antialias: true });
renderer.setSize(window.innerWidth, window.innerHeight);
renderer.shadowMap.enabled = true;
renderer.shadowMap.type = THREE.PCFSoftShadowMap;
document.body.appendChild(renderer.domElement);

const controls = new OrbitControls(camera, renderer.domElement);
controls.target.set(0, 2.3, 0);
controls.enableDamping = true;

// 2. Studio Lighting
const ambient = new THREE.AmbientLight(0xd9e5f3, 0.75);
scene.add(ambient);

const keyLight = new THREE.DirectionalLight(0xffffff, 1.4);
keyLight.position.set(9, 14, 11);
keyLight.castShadow = true;
scene.add(keyLight);

const fillLight = new THREE.DirectionalLight(0x93c5fd, 0.6);
fillLight.position.set(-10, 8, -6);
scene.add(fillLight);

// 3. Materials
// Solid CAD material (Image 2)
const matSolid = new THREE.MeshStandardMaterial({
  color: 0x98a4b3,
  roughness: 0.32,
  metalness: 0.55,
  side: THREE.DoubleSide
});

// CAD Edges line material (Image 1)
const matLine = new THREE.LineBasicMaterial({
  color: 0x18202a,
  linewidth: 1.5
});

// Helper to create hollow tank geometry with wall thickness & bottom plate
function createHollowTank(outerR, innerR, height, bottomThick = 0.08) {
  const points = [
    new THREE.Vector2(0, bottomThick),
    new THREE.Vector2(innerR, bottomThick),
    new THREE.Vector2(innerR, height),
    new THREE.Vector2(outerR, height),
    new THREE.Vector2(outerR, 0.05),
    new THREE.Vector2(outerR - 0.05, 0),
    new THREE.Vector2(0, 0)
  ];
  return new THREE.LatheGeometry(points, 56);
}

// Helper to attach crisp CAD edges to any mesh
function addCadEdges(mesh, threshold = 20) {
  const edges = new THREE.EdgesGeometry(mesh.geometry, threshold);
  const line = new THREE.LineSegments(edges, matLine);
  mesh.add(line);
  return mesh;
}

const assembly = new THREE.Group();

// A. Center Tank (radius 1.25, height 3.2, base at Y=0.6)
const centerGeom = createHollowTank(1.25, 1.17, 3.2, 0.1);
const centerMesh = addCadEdges(new THREE.Mesh(centerGeom, matSolid));
centerMesh.position.set(0, 0.6, 0);
assembly.add(centerMesh);

// Center Tank 4 Base Feet (height 0.6)
const footGeom = new THREE.CylinderGeometry(0.08, 0.08, 0.6, 24);
footGeom.translate(0, -0.3, 0);
[0, Math.PI * 0.5, Math.PI, Math.PI * 1.5].forEach(angle => {
  const foot = addCadEdges(new THREE.Mesh(footGeom, matSolid));
  foot.position.set(Math.cos(angle) * 0.9, 0.6, Math.sin(angle) * 0.9);
  assembly.add(foot);
});

// Center Tank Lower Discharge Pipes (Left & Right)
const lowerPipeGeom = new THREE.CylinderGeometry(0.11, 0.11, 1.25, 24);
lowerPipeGeom.rotateZ(Math.PI / 2);
lowerPipeGeom.translate(0.625, 0, 0);

const lowerLeft = addCadEdges(new THREE.Mesh(lowerPipeGeom, matSolid));
lowerLeft.position.set(-1.25, 1.2, 0);
lowerLeft.rotation.y = Math.PI; // points left
assembly.add(lowerLeft);

const lowerRight = addCadEdges(new THREE.Mesh(lowerPipeGeom, matSolid));
lowerRight.position.set(1.25, 1.2, 0); // points right
assembly.add(lowerRight);

// B. Side Elevated Tanks (Left: X=-3.8, Right: X=+3.8, base at Y=3.0)
const sideGeom = createHollowTank(1.0, 0.93, 2.2, 0.08);
const legGeom = new THREE.CylinderGeometry(0.065, 0.065, 3.0, 24);
legGeom.translate(0, -1.5, 0);

[-3.8, 3.8].forEach(posX => {
  // Elevated Tank
  const tank = addCadEdges(new THREE.Mesh(sideGeom, matSolid));
  tank.position.set(posX, 3.0, 0);
  assembly.add(tank);

  // 4 Slender Support Legs
  [Math.PI * 0.25, Math.PI * 0.75, Math.PI * 1.25, Math.PI * 1.75].forEach(ang => {
    const leg = addCadEdges(new THREE.Mesh(legGeom, matSolid));
    leg.position.set(posX + Math.cos(ang) * 0.74, 3.0, Math.sin(ang) * 0.74);
    assembly.add(leg);
  });
});

// C. Upper Connecting Pipes with 90° Smooth Elbows
function createPipe(isLeft) {
  const sign = isLeft ? -1 : 1;
  const dropX = sign * 3.15;
  const inletY = 2.9;
  const bendR = 0.35;
  const targetX = sign * 1.25;

  const path = new THREE.CurvePath();
  path.add(new THREE.LineCurve3(new THREE.Vector3(dropX, 3.0, 0), new THREE.Vector3(dropX, inletY + bendR, 0)));
  path.add(new THREE.QuadraticBezierCurve3(
    new THREE.Vector3(dropX, inletY + bendR, 0),
    new THREE.Vector3(dropX, inletY, 0),
    new THREE.Vector3(dropX - sign * bendR, inletY, 0)
  ));
  path.add(new THREE.LineCurve3(new THREE.Vector3(dropX - sign * bendR, inletY, 0), new THREE.Vector3(targetX, inletY, 0)));

  const tube = addCadEdges(new THREE.Mesh(new THREE.TubeGeometry(path, 64, 0.11, 24, false), matSolid));
  assembly.add(tube);

  // Collar Sleeve
  const collar = addCadEdges(new THREE.Mesh(new THREE.CylinderGeometry(0.15, 0.15, 0.32, 24), matSolid));
  collar.geometry.rotateZ(Math.PI / 2);
  collar.position.set(sign * 2.2, inletY, 0);
  assembly.add(collar);
}

createPipe(true);  // Left pipe
createPipe(false); // Right pipe

scene.add(assembly);

// Render loop
function animate() {
  requestAnimationFrame(animate);
  controls.update();
  renderer.render(scene, camera);
}
animate();

window.addEventListener('resize', () => {
  camera.aspect = window.innerWidth / window.innerHeight;
  camera.updateProjectionMatrix();
  renderer.setSize(window.innerWidth, window.innerHeight);
});
